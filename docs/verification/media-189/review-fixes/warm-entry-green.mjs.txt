import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { dirname, resolve, extname } from 'node:path';
import { createRequire } from 'node:module';
import vm from 'node:vm';
import ts from 'typescript';
import { QueryClient, QueryObserver } from '@tanstack/react-query';

const external = createRequire(resolve('package.json'));
const jsx = (type, props) => ({ type, props });
const ui = new Proxy({}, { get: (_object, name) => String(name) });
const checks = [];
function scenario() {
  const client = new QueryClient();
  const pageSlots = [], editorSlots = [];
  let slots = pageSlots, cursor = 0, changed = false, observer;
  let unlisten;
  const reads = [], writes = [];
  const modules = new Map();
  const hooks = {
    useState(initial) {
      const currentSlots = slots, index = cursor++;
      if (!(index in currentSlots)) currentSlots[index] = typeof initial === 'function' ? initial() : initial;
      return [currentSlots[index], next => {
        const value = typeof next === 'function' ? next(currentSlots[index]) : next;
        if (!Object.is(value, currentSlots[index])) changed = true;
        currentSlots[index] = value;
      }];
    },
    useRef(initial) {
      const index = cursor++;
      if (!(index in slots)) slots[index] = { current: initial };
      return slots[index];
    },
    useCallback: callback => callback,
  };
  const fetch = (url, init) => {
    if (init?.method === 'PATCH') {
      const body = JSON.parse(init.body);
      writes.push({ url, body });
      return Promise.resolve({ ok: true, json: async () => ({ ...body, id: 1, updatedAt: 'saved' }) });
    }
    return new Promise((resolve, reject) => reads.push({ url, resolve: data => resolve({ ok: true, json: async () => data }), reject }));
  };
  function load(file) {
    if (modules.has(file)) return modules.get(file);
    const exports = {};
    modules.set(file, exports);
    const require = id => {
      if (id === 'react/jsx-runtime') return { jsx, jsxs: jsx, Fragment: 'Fragment' };
      if (id === 'react') return hooks;
      if (id === '@tanstack/react-query') return {
        useQueryClient: () => client,
        useQuery(options) {
          if (!observer) observer = new QueryObserver(client, options);
          else observer.setOptions(options);
          return observer.getCurrentResult();
        },
      };
      if (id === '@heroui/react/toast') return { toast: () => {} };
      if (id === '@heroui/react/alert-dialog') return { AlertDialog: ui };
      if (id.startsWith('@heroui/react/')) return ui;
      if (id === '../upload/provider') return { useUploadQueue: () => ({ client: { invalidateQueries: async () => {} } }) };
      if (['../shell/owner-shell', '../shell/settings-categories', './settings-form', './preview-pane', './use-preview'].includes(id)) return ui;
      if (!id.startsWith('.')) return external(id);
      return load(resolve(dirname(file), extname(id) ? id : `${id}.ts`));
    };
    const code = ts.transpileModule(readFileSync(file, 'utf8'), { compilerOptions: { module: ts.ModuleKind.CommonJS, jsx: ts.JsxEmit.ReactJSX } }).outputText;
    vm.runInNewContext(code, { exports, require, fetch, requestAnimationFrame: () => {}, console });
    return exports;
  }
  const { initialMediaSettings } = load(resolve('src/server/media/validation.ts'));
  const cached = { ...initialMediaSettings, id: 1, updatedAt: 'cached', quality: 82 };
  client.setQueryData(['processing-settings'], cached);
  const { ProcessingPage } = load(resolve('src/components/processing/processing-page.tsx'));
  const { useProcessingSettings } = load(resolve('src/components/processing/use-processing-settings.ts'));
  const page = () => {
    let tree;
    for (let n = 0; n < 5; n++) {
      slots = pageSlots; cursor = 0; changed = false;
      tree = ProcessingPage({});
      if (!changed) return tree;
    }
    throw new Error('Production render failed to settle');
  };
  const editor = initial => { slots = editorSlots; cursor = 0; return useProcessingSettings(initial); };
  const start = () => { const tree = page(); unlisten = observer.subscribe(() => {}); return tree; };
  return { client, reads, writes, cached, page, editor, start, close() { unlisten(); client.clear(); } };
}
function nodes(tree) {
  if (Array.isArray(tree)) return tree.flatMap(nodes);
  if (!tree?.props) return [];
  return [tree, ...nodes(tree.props.children)];
}
const state = tree => nodes(tree).find(node => node.props['data-testid'] === 'processing-editor')?.props['data-state'];
const isEditor = tree => tree?.type?.name === 'ProcessingEditor';
const settle = () => new Promise(resolve => setTimeout(resolve, 0));

for (const failFirst of [false, true]) {
  const env = scenario();
  try {
    let tree = env.start();
    assert.equal(state(tree), 'loading');
    assert.equal(isEditor(tree), false);
    assert.equal(env.reads.length, 1);
    tree = env.page();
    assert.equal(state(tree), 'loading');
    assert.equal(isEditor(tree), false);
    if (failFirst) {
      env.reads[0].reject(new TypeError('isolated initial GET failure'));
      await settle();
      tree = env.page();
      assert.equal(state(tree), 'error');
      assert.equal(isEditor(tree), false);
      const retry = nodes(tree).find(node => node.props['data-testid'] === 'processing-settings-retry');
      assert.ok(retry);
      retry.props.onPress();
      tree = env.page();
      assert.equal(state(tree), 'loading');
      assert.equal(env.reads.length, 2);
    }
    env.reads.at(-1).resolve({ ...env.cached, quality: 68, updatedAt: 'fresh' });
    await settle();
    tree = env.page();
    assert.equal(isEditor(tree), true);
    assert.equal(tree.props.initial.quality, 68);
    const initial = tree.props.initial;
    let form = env.editor(initial);
    await form.save();
    assert.equal(env.writes.length, 1);
    assert.equal(env.writes[0].body.quality, 68);
    assert.equal(Object.keys(env.writes[0].body).length, 20);
    form.change('quality', 63);
    form = env.editor(initial);
    assert.equal(form.input.quality, 63);
    env.client.setQueryData(['processing-settings'], { ...initial, quality: 71 });
    tree = env.page();
    assert.equal(isEditor(tree), true);
    assert.equal(tree.props.initial, initial);
    assert.equal(tree.key, undefined);
    form = env.editor(tree.props.initial);
    assert.equal(form.input.quality, 63);
    assert.equal(env.reads.length, failFirst ? 2 : 1);
    checks.push({ firstRead: failFirst ? 'failure -> explicit retry' : 'success', pendingState: 'loading without editor', freshQuality: 68, savedQuality: env.writes[0].body.quality, savedFields: Object.keys(env.writes[0].body).length, changedCacheQuality: 71, retainedDraftQuality: form.input.quality, retainedEditorInitialIdentity: true });
  } finally { env.close(); }
}
console.log(JSON.stringify({ scope: 'Installed real TanStack QueryObserver and current transpiled production ProcessingPage/useProcessingSettings. Isolated persistent hook slots, JSX and deferred fetch; no ReactDOM, browser or real HTTP.', checks, status: 'passed' }, null, 2));
